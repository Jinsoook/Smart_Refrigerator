핵심 로직 — days_left() : 유통기한 계산
def days_left(self) -> int:
    db_entry = EXPIRY_DB.get(self.name)
    if db_entry is None:
        return 999          # DB 미등록 → 유통기한 미적용

    expiry_date = self.purchase_date + timedelta(days=db_entry["days"])
    remaining   = (expiry_date - date.today()).days
    return remaining        # 음수 = 이미 만료됨

def use_ingredient(self, name: str, qty: float) -> None:
    if name not in self.items:           raise KeyError(...)
    if qty <= 0:                         raise ValueError(...)
    if qty > self.items[name].quantity:  raise ValueError(...)

    self.items[name].quantity -= qty

    self.history.append({               # 사용 이력 기록
        "action": "used", "name": name,
        "qty": qty, "date": date.today().isoformat(),
    })

    if self.items[name].quantity <= 0:  # 수량 0 → 자동 삭제
        del self.items[name]
Engine RecipeRecommender recommender.py
3개의 서브 메트릭(보유 비율·임박도·부족 패널티)을 가중 합산해 레시피를 점수화하는 추천 엔진. stateless 클래스(인스턴스 속성 없음)로 설계되어 여러 뷰에서 재사용됩니다.

추천 점수 공식 (score formula)
score = 0.5 × match_ratio + 0.4 × urgency - 0.1 × missing_penalty match_ratio = 냉장고 보유 재료 수 / 레시피 전체 재료 수 (0~1) urgency = Σ max(0, 1 - days_left_i / 7) / 전체 재료 수 → 7일 이내 임박할수록 urgency_i → 1.0 → 만료 재료(days_left < 0)는 urgency_i = 0 missing_penalty = 부족 재료 수 / 레시피 전체 재료 수 (0~1)
def score(self, recipe, fridge, w1=0.5, w2=0.4, w3=0.1) -> float:
    match   = self._match_ratio(recipe, fridge)
    urgency = self._urgency(recipe, fridge)
    penalty = self._missing_penalty(recipe, fridge)
    return w1 * match + w2 * urgency - w3 * penalty

def recommend(self, fridge, top_n=3) -> list[tuple]:
    scored = [(r, self.score(r, fridge)) for r in RECIPE_DB]
    return sorted(scored, key=lambda x: x[1], reverse=True)[:top_n]

def show_view(self, name: str) -> None:
    for view in self.views.values():
        view.grid_remove()          # 레이아웃 정보 보존하며 숨김

    self.views[name].grid()         # 선택된 뷰만 표시
    self.views[name].refresh()      # 최신 Fridge 데이터로 UI 갱신
    self.current_view = name

    for k, btn in self._nav_buttons.items():
        btn.configure(              # 활성 버튼 강조
            fg_color=("gray75","gray25") if k == name else "transparent"
        )

def _refresh_charts(self) -> None:
    # 기존 캔버스·Figure 해제 (메모리 누수 방지)
    if self._canvas is not None:
        self._canvas.get_tk_widget().destroy()
        self._canvas = None
    if self._fig is not None:
        plt.close(self._fig)        # matplotlib Figure 메모리 해제
        self._fig = None

    # 새 Figure 생성 (2-subplot: 막대 + 파이)
    self._fig = Figure(figsize=(10, 3.8), dpi=90, facecolor="#FFFFFF")
    ax1 = self._fig.add_subplot(1, 2, 1)   # 사용 빈도 Top 5
    ax2 = self._fig.add_subplot(1, 2, 2)   # 카테고리 파이

    # TkAgg 캔버스로 CTkFrame에 임베딩
    self._canvas = FigureCanvasTkAgg(self._fig, master=self._chart_frame)
    self._canvas.draw()
    self._canvas.get_tk_widget().grid(row=0, column=0, sticky="nswe")
